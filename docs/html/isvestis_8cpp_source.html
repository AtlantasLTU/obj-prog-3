50int lietuviskosRaides(const std::string& eilute)
51{
52 int simboliuKiekis = 0;
53 for (char c : eilute)
54 {
55 // jei baitas neprasideda su 10xxxxxx, tai naujas simbolis
56 if ((c & 0xC0) != 0x80)
57 { // paprastas ASCII simbolis prasideda su 0, keliu baitu pvz lietuviskos raides prasideda su 11 arba 111 arba 1111, priklausomai nuo kodavimo | 0xC0 = 11000000, 0x80 = 10000000. & (AND) bit'u operacija atranda ar c prasideda su 0 ar 1. antras, trecias ar ketvirtas baitas UTF-8 kodavime visad prasides su 10xxxxxx

59 }
60 }
