886 void reallocate(size_type new_cap) {
887 pointer new_data = std::allocator_traits<Allocator>::allocate(alloc_, new_cap);
888 size_type i = 0;
889 try {
890 for (; i < size_; ++i) {
891 std::allocator_traits<Allocator>::construct(alloc_, new_data + i,
892 std::move_if_noexcept(*(data_ + i)));
893 }
894 } catch (...) {
895 for (size_type j = 0; j < i; ++j) {
896 std::allocator_traits<Allocator>::destroy(alloc_, new_data + j);
897 }
898 std::allocator_traits<Allocator>::deallocate(alloc_, new_data, new_cap);
899 throw;
900 }
901 for (size_type i = 0; i < size_; ++i) {
902 std::allocator_traits<Allocator>::destroy(alloc_, data_ + i);
903 }
904 if (data_) {
905 std::allocator_traits<Allocator>::deallocate(alloc_, data_, capacity_);
906 }
907 data_ = new_data;
908 capacity_ = new_cap;
909 }
